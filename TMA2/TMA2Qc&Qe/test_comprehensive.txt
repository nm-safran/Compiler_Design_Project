};

class Engine {
    private attribute power: float;
    private attribute running: integer;

    public constructor(p: float) {
        self.power := p;
        self.running := 0;
    }

    public func start() => integer {
        if (self.power > 0.0) then {
            self.running := 1;
            return (1);
        } else {
            return (0);
        };
    }

    public func stop() => void {
        self.running := 0;
    }
};

implement Vehicle {
    func start() => void {

    }

    func addPassenger(person: Person) => void {
        local count: integer;
        local i: integer;

        count := 0;
        i := 0;

        while (i < 10) {
            if (not (self.passengers[i] == person)) then {
                count := count + 1;
            };
            i := i + 1;
        };

        self.passengers[count] := person;
    }
}
