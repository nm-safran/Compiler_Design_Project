// Test File 6: test_expressions.txt
// Tests complex expressions and operators
func testExpressions() => void {
    local x: integer;
    local y: float;
    local result: integer;

    x := 10 + 5 * 3 - 2;
    y := 3.14 * 2.0 + 1.5;

    if (x == 23 and y > 8.0) then {
        result := x * 2;
    } else {
        result := x / 2;
    };

    if (not (x <> y) or x < 100) then {
        write(result);
    };
}
